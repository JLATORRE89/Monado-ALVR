cmake_minimum_required(VERSION 3.16)
project(intel_xr_checkerboard C)
set(CMAKE_C_STANDARD 11)
find_package(Vulkan REQUIRED)
find_path(OPENXR_INCLUDE_DIR openxr/openxr.h HINTS /usr/include /usr/local/include)
find_library(OPENXR_LOADER openxr_loader HINTS /usr/lib /usr/local/lib /usr/lib/x86_64-linux-gnu)
if(NOT OPENXR_INCLUDE_DIR OR NOT OPENXR_LOADER)
  message(FATAL_ERROR "OpenXR headers/loader not found")
endif()
add_executable(intel_xr_checkerboard main.c)
target_include_directories(intel_xr_checkerboard PRIVATE ${OPENXR_INCLUDE_DIR})
target_link_libraries(intel_xr_checkerboard PRIVATE ${OPENXR_LOADER} Vulkan::Vulkan m)
