cmake_minimum_required(VERSION 3.16)
project(intel_xr_checkerboard C)
set(CMAKE_C_STANDARD 11)
find_package(Vulkan REQUIRED)
set(OPENXR_SDK_SOURCE "/ai/intel-xr-prototype/src/OpenXR-SDK-Source" CACHE PATH "OpenXR SDK source")
set(OPENXR_SDK_BUILD "/ai/intel-xr-prototype/build/openxr-demo" CACHE PATH "OpenXR SDK build")
find_path(OPENXR_INCLUDE_DIR openxr/openxr.h
  HINTS "${OPENXR_SDK_SOURCE}/include" /usr/include /usr/local/include)
find_library(OPENXR_LOADER openxr_loader
  HINTS "${OPENXR_SDK_BUILD}/src/loader" "${OPENXR_SDK_BUILD}/src/loader/RelWithDebInfo"
        /usr/lib /usr/local/lib /usr/lib/x86_64-linux-gnu)
if(NOT OPENXR_INCLUDE_DIR OR NOT OPENXR_LOADER)
  message(FATAL_ERROR "OpenXR headers/loader not found. Build hello_xr first with scripts/build-openxr-demo.sh")
endif()
message(STATUS "OpenXR include: ${OPENXR_INCLUDE_DIR}")
message(STATUS "OpenXR loader:  ${OPENXR_LOADER}")
add_executable(intel_xr_checkerboard main.c)
target_include_directories(intel_xr_checkerboard PRIVATE ${OPENXR_INCLUDE_DIR})
target_link_libraries(intel_xr_checkerboard PRIVATE ${OPENXR_LOADER} Vulkan::Vulkan m)
set_target_properties(intel_xr_checkerboard PROPERTIES BUILD_RPATH "${OPENXR_SDK_BUILD}/src/loader")
